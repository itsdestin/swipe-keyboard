if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(swipekb_config_probe LANGUAGES CXX)

install(TARGETS swipekb-config RUNTIME DESTINATION share/swipe-keyboard/integration)
